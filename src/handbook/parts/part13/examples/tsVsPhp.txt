<?php
// PHP 7 type hints: checked at RUNTIME, when the line actually runs
function formatRupiah(int $n): string
{
    return 'Rp ' . number_format($n, 0, ',', '.');
}

formatRupiah('abc'); // TypeError... but only when this code is executed
