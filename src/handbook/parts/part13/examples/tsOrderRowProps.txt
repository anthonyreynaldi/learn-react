// OrderRow.tsx: typed props
type OrderRowProps = {
  order: Order
  onApprove: (id: number) => void
  highlighted?: boolean
}

export function OrderRow({ order, onApprove, highlighted = false }: OrderRowProps) {
  return (
    <tr style={{ fontWeight: highlighted ? 'bold' : undefined }}>
      <td>#{order.id}</td>
      <td>{order.customer}</td>
      <td>
        <button onClick={() => onApprove(order.id)}>Approve</button>
      </td>
    </tr>
  )
}
