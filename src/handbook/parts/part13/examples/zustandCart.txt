// npm i zustand   (not installed in this project)
// cartStore.js: a global store, no Provider needed
import { create } from 'zustand'

export const useCartStore = create((set) => ({
  items: [],

  // Actions update the state immutably, like a reducer (Part 10)
  addItem: (product) =>
    set((state) => {
      const existing = state.items.find((item) => item.name === product.name)
      return {
        items: existing
          ? state.items.map((item) => (item.name === product.name ? { ...item, qty: item.qty + 1 } : item))
          : [...state.items, { ...product, qty: 1 }],
      }
    }),
  removeItem: (name) => set((state) => ({ items: state.items.filter((item) => item.name !== name) })),
  clear: () => set({ items: [] }),
}))

// Any component, anywhere: SELECT only what it needs.
// It re-renders only when that selected value changes.
function CartBadge() {
  const count = useCartStore((state) => state.items.reduce((sum, item) => sum + item.qty, 0))
  return <span>🛒 {count}</span>
}

function AddButton({ product }) {
  const addItem = useCartStore((state) => state.addItem)
  return <button onClick={() => addItem(product)}>Add</button>
}
