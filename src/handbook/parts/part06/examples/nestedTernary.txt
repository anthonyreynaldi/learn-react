// EASIER: a lookup object, a variable with if/else, or a small component
function OrderActions({ status, isPaid }) {
  if (status === 'pending') return <button>Approve</button>
  if (status === 'approved') return isPaid ? <span>Paid</span> : <button>Print invoice</button>
  if (status === 'rejected') return null
  return <span>Unknown</span>
}
