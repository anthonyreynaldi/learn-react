// ✗ NESTED: customers contain orders, orders contain lines.
// Updating one line's qty means copying three levels:
const [customers, setCustomers] = useState([
  {
    id: 1, name: 'PT Maju Jaya',
    orders: [
      { id: 101, status: 'pending', lines: [{ id: 'a', product: 'Floor Tile 60x60', qty: 3 }] },
    ],
  },
])

setCustomers(customers.map((c) => c.id !== 1 ? c : {
  ...c,
  orders: c.orders.map((o) => o.id !== 101 ? o : {
    ...o,
    lines: o.lines.map((l) => l.id !== 'a' ? l : { ...l, qty: 4 }),
  }),
}))
