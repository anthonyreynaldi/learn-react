// Zod: the same rules as a schema on the CLIENT (Part 13; not installed here)
import { z } from 'zod'

const orderSchema = z.object({
  customer: z.string().trim().min(1, 'Please enter a customer.').max(100),
  warehouse: z.enum(['SBY', 'JKT']),
  deliveryDate: z.string().date(), // "YYYY-MM-DD"
  lines: z
    .array(
      z.object({
        product: z.string().min(1),
        qty: z.coerce.number().int().min(1), // "3" → 3
      }),
    )
    .min(1),
})
