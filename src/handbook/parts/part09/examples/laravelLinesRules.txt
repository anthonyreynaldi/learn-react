<?php
// Validating an array of order lines in Laravel 5
$this->validate($request, [
    'customer'        => 'required|max:100',
    'lines'           => 'required|array|min:1',
    // The * means "every item in the array"
    'lines.*.product' => 'required',
    'lines.*.qty'     => 'required|integer|min:1',
]);
