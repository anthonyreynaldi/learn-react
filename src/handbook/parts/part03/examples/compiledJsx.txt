// What you write:
const element = <h1 className="title">Hello {name}</h1>

// What the compiler (Vite / Babel) turns it into:
import { jsx as _jsx } from 'react/jsx-runtime'

const element = _jsx('h1', {
  className: 'title',
  children: ['Hello ', name],
})

// _jsx(...) returns a plain object, roughly:
// { type: 'h1', props: { className: 'title', children: ['Hello ', 'PT Maju Jaya'] } }
