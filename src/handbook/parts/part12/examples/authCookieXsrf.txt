// SESSION COOKIE auth (same domain: islands, or an SPA served by Laravel).
// The browser sends the laravel_session cookie automatically. For POST/PUT/DELETE,
// Laravel's VerifyCsrfToken middleware also wants the CSRF token. Laravel puts it
// in a readable XSRF-TOKEN cookie; we copy it into the X-XSRF-TOKEN header.

function getCookie(name) {
  const match = document.cookie.match(new RegExp('(^|;\\s*)' + name + '=([^;]*)'))
  return match ? decodeURIComponent(match[2]) : null
}

async function postJson(url, body) {
  const response = await fetch(url, {
    method: 'POST',
    credentials: 'same-origin', // send the session cookie
    headers: {
      'Content-Type': 'application/json',
      Accept: 'application/json',
      'X-XSRF-TOKEN': getCookie('XSRF-TOKEN'), // Laravel decrypts and checks it
    },
    body: JSON.stringify(body),
  })
  return response.json()
}
