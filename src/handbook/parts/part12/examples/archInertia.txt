use Inertia\Inertia;

class OrderController extends Controller
{
    public function index()
    {
        // Instead of view('orders.index', [...]):
        // render the React page component resources/js/Pages/Orders/Index.jsx
        // and pass the data as its props.
        return Inertia::render('Orders/Index', [
            'orders' => Order::latest()->get(),
        ]);
    }
}
